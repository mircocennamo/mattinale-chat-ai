function sendMessage() {
const message = userInput.value.trim();
if (message === '') return;

// Aggiungi il messaggio dell'utente alla chat
addMessageToChat('user', message);
userInput.value = '';

// Mostra l'indicatore di digitazione
showTypingIndicator();

// Invia la richiesta al backend
fetch('http://localhost:9999/rag/ask?question=' + encodeURIComponent(message), {
method: 'GET',
headers: {
    'Accept': '*/*'
}
})
.then(response => response.text())
.then(data => {
// Rimuovi l'indicatore di digitazione
removeTypingIndicator();
// Aggiungi la risposta dell'AI alla chat
addMessageToChat('ai', data);
})
.catch(error => {
console.error('Errore:', error);
removeTypingIndicator();
addMessageToChat('ai', 'Mi dispiace, si è verificato un errore. Riprova più tardi.');
});
}

function showTypingIndicator() {

}

function addMessageToChat(sender, content) {
    const messageDiv = document.createElement('div');
    
    if (sender === 'user') {
        messageDiv.className = 'message user-message';
    } else {
        messageDiv.className = 'message ai-message';
    }
    
    // Formatta la data in italiano
    const now = new Date();
    const options = { hour: '2-digit', minute: '2-digit' };
    const formattedTime = now.toLocaleTimeString('it-IT', options);
    
    // Aggiungi intestazione, contenuto e metadati
    messageDiv.innerHTML = `
        <div class="message-header">
            <i class="fas ${sender === 'user' ? 'fa-user' : 'fa-robot'}"><\/i>
            ${sender === 'user' ? 'Domanda' : 'Risposta'}
        <\/div>
        <div class="message-content">${content}<\/div>
        <div class="message-meta">${sender === 'user' ? 'Tu' : 'AI'}, oggi alle ${formattedTime}<\/div>
    `;
    
    chatContainer.appendChild(messageDiv);
    chatContainer.scrollTop = chatContainer.scrollHeight;
    
    // Salva il messaggio nella cronologia locale
    saveMessageToLocalStorage(sender, content);
}

function saveMessageToLocalStorage(sender, content) {
let chatHistory = JSON.parse(localStorage.getItem('chatHistory') || '[]');
chatHistory.push({
sender: sender,
content: content,
timestamp: new Date().toISOString()
});
// Limita la cronologia a 50 messaggi
if (chatHistory.length > 50) {
chatHistory = chatHistory.slice(chatHistory.length - 50);
}
localStorage.setItem('chatHistory', JSON.stringify(chatHistory));
}

function loadChatHistory() {
    // Prima controlla se c'è una cronologia locale
    const chatHistory = JSON.parse(localStorage.getItem('chatHistory') || '[]');
    
    if (chatHistory.length > 0) {
        chatContainer.innerHTML = '';
        
        // Aggiungi un'intestazione per lo storico della chat
        const historyHeader = document.createElement('div');
        historyHeader.className = 'message-separator';
        historyHeader.textContent = 'Storico della conversazione';
        chatContainer.appendChild(historyHeader);
        
        chatHistory.forEach((msg, index) => {
            const messageDiv = document.createElement('div');
            
            if (msg.sender === 'user') {
                messageDiv.className = 'message user-message';
            } else {
                messageDiv.className = 'message ai-message';
            }
            
            // Formatta la data in italiano
            const msgDate = new Date(msg.timestamp);
            const options = { hour: '2-digit', minute: '2-digit' };
            const formattedTime = msgDate.toLocaleTimeString('it-IT', options);
            
            // Aggiungi intestazione, contenuto e metadati
            messageDiv.innerHTML = `
                <div class="message-header">
                    <i class="fas ${msg.sender === 'user' ? 'fa-user' : 'fa-robot'}"><\/i>
                    ${msg.sender === 'user' ? 'Domanda' : 'Risposta'}
                <\/div>
                <div class="message-content">${msg.content}<\/div>
                <div class="message-meta">${msg.sender === 'user' ? 'Tu' : 'AI'}, ${formattedTime}<\/div>
            `;
            
            chatContainer.appendChild(messageDiv);
            
            // Aggiungi un separatore tra le coppie domanda-risposta
            if (msg.sender === 'ai' && index < chatHistory.length - 1) {
                const separator = document.createElement('div');
                separator.className = 'message-separator';
                chatContainer.appendChild(separator);
            }
        });
    } else {
        // Aggiungi un messaggio di benvenuto se non c'è cronologia
        const welcomeMessage = document.createElement('div');
        welcomeMessage.className = 'message ai-message';
        welcomeMessage.innerHTML = `
            <div class="message-header">
                <i class="fas fa-robot"><\/i>
                Benvenuto
            <\/div>
            <div class="message-content">Benvenuto! Fammi domande sui mattinali caricati.<\/div>
            <div class="message-meta">Sistema, oggi alle ${new Date().toLocaleTimeString('it-IT', { hour: '2-digit', minute: '2-digit' })}<\/div>
        `;
        chatContainer.appendChild(welcomeMessage);
    }
    
    chatContainer.scrollTop = chatContainer.scrollHeight;
}

function uploadExcelFile() {
const fileInput = document.getElementById('excelFile');
const file = fileInput.files[0];

if (!file) {
uploadExcelStatus.className = 'alert alert-danger';
uploadExcelStatus.style.display = 'block';
uploadExcelStatus.textContent = 'Seleziona un file da caricare.';
return;
}

const formData = new FormData();
formData.append('file', file);

uploadExcelStatus.className = 'alert alert-info';
uploadExcelStatus.style.display = 'flex';
uploadExcelStatus.innerHTML = `
<div class="spinner"><\/div>
<span>Caricamento in corso...<\/span>
`;

fetch('http://localhost:9999/uploadExcel', {
method: 'POST',
body: formData
})
.then(response => {
if (!response.ok) {
    throw new Error('Errore nel caricamento del file');
}
return response.text();
})
.then(data => {
uploadExcelStatus.className = 'alert alert-success';
uploadExcelStatus.style.display = 'block';
uploadExcelStatus.textContent = 'File caricato con successo!';
fileInput.value = '';

// Aggiungi un messaggio di sistema nella chat
addMessageToChat('ai', `File Excel "${file.name}" caricato con successo. Ora puoi fare domande sui dati contenuti nel file.`);

// Passa automaticamente alla tab della chat
document.querySelector('[data-tab="chat"]').click();
})
.catch(error => {
console.error('Errore:', error);
uploadExcelStatus.className = 'alert alert-danger';
uploadExcelStatus.style.display = 'block';
uploadExcelStatus.textContent = 'Si è verificato un errore durante il caricamento del file.';
});
}

  <script>
    // ===== Configurazione aggancio =====
    const SELECTOR_HOOK = '#chat, .chat, [data-role="chat"]';

    // ===== Utilità Base64 =====
    function extractDataImageURLs(text){
      if(!text) return [];
      const re = /data:image\/(png|jpeg|jpg|gif|webp|svg\+xml);base64,[A-Za-z0-9+/=]+/g;
      const set = new Set();
      let m; while ((m = re.exec(text)) !== null){ set.add(m[0]); }
      return Array.from(set);
    }

    function render(urls){
      const gallery = document.getElementById('gallery');
      const wrap = document.getElementById('b64wrap');
      gallery.innerHTML = '';
      urls.forEach((u, i)=>{
        const card = document.createElement('div'); card.className='card';
        const img = document.createElement('img'); img.src = u; img.alt = `immagine ${i+1}`; img.loading='lazy';
        const mime = u.substring(5, u.indexOf(';'));
        const sizeBytes = Math.floor((u.length - u.indexOf(',')) * 3/4);
        const kb = (sizeBytes/1024).toFixed(1);
        const meta = document.createElement('div'); meta.className='meta'; meta.textContent = `${mime} • ~${kb} KB`;
        card.appendChild(img); card.appendChild(meta);
        gallery.appendChild(card);
      });
      wrap.style.display = urls.length ? 'block' : 'none';
    }

    // ===== Scansione iniziale =====
    function scanAll(){
      const targets = Array.from(document.querySelectorAll(SELECTOR_HOOK));
      let combined = '';
      if(targets.length){
        targets.forEach(t=> combined += '
' + (t.innerText || t.textContent || ''));
      } else {
        combined = document.body.innerText || document.body.textContent || '';
      }
      // includi anche il blocco originale
      combined += '
' + (document.getElementById('original').textContent || '');
      const urls = extractDataImageURLs(combined);
      render(urls);
    }

    scanAll();

    // ===== Osservatori =====
    const opts = {childList:true, subtree:true, characterData:true};

    // osserva i target definiti
    const observer = new MutationObserver(()=> scanAll());
    const hookNodes = document.querySelectorAll(SELECTOR_HOOK);
    if(hookNodes.length){
      hookNodes.forEach(n=> observer.observe(n, opts));
    } else {
      // fallback: osserva il body
      observer.observe(document.body, opts);
    }

    // osserva anche il blocco originale (in caso venga aggiornato via script)
    const originalNode = document.getElementById('original');
    if(originalNode){
      const obs2 = new MutationObserver(()=> scanAll());
      obs2.observe(originalNode, opts);
    }
